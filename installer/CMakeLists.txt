cmake_minimum_required(VERSION 3.20)
project(GlazeInstaller C)
set(CMAKE_C_STANDARD 11)
find_package(PkgConfig REQUIRED)
pkg_check_modules(USB REQUIRED IMPORTED_TARGET libusb-1.0)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
set(U ${ROOT}/host/glaze_usb)
add_executable(glaze-installer-usb helper.c ${U}/transport.c ${U}/client.c ${U}/pcconnect.c ${U}/nand.c ${U}/nand_write.c)
target_include_directories(glaze-installer-usb PRIVATE compat ${U} ${ROOT}/lib/include)
target_link_libraries(glaze-installer-usb PRIVATE PkgConfig::USB)
add_executable(nand-inventory ${ROOT}/host/nand_inventory.c ${ROOT}/kernel/drivers/nand_boot.c ${ROOT}/kernel/drivers/nand_ecc.c)
file(GLOB DHARA ${ROOT}/third_party/dhara/dhara/*.c ${ROOT}/third_party/dhara/ecc/*.c)
add_executable(mkfsimg ${ROOT}/host/mkfsimg.c ${ROOT}/servers/fs/ftl.c ${ROOT}/third_party/littlefs/lfs.c ${ROOT}/third_party/littlefs/lfs_util.c ${DHARA})
target_include_directories(mkfsimg PRIVATE ${ROOT}/servers/fs ${ROOT}/third_party/littlefs ${ROOT}/third_party/dhara)
target_compile_definitions(mkfsimg PRIVATE LFS_NO_DEBUG LFS_NO_WARN LFS_NO_ERROR)
foreach(t glaze-installer-usb nand-inventory mkfsimg)
  target_compile_options(${t} PRIVATE -Wall -Wextra -include ${CMAKE_CURRENT_SOURCE_DIR}/compat.h)
  if(WIN32)
    target_link_libraries(${t} PRIVATE bcrypt)
  endif()
endforeach()
